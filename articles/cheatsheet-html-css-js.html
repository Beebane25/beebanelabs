📦 CSS Flexbox

/* Container */
display: flex;
flex-direction: row | column;
justify-content: flex-start | center | space-between | space-around;
align-items: stretch | center | flex-start | flex-end;
flex-wrap: wrap | nowrap;
gap: 16px;

/* Item */
flex: 1;              /* grow:1, shrink:1, basis:0 */
flex: 0 0 300px;      /* no grow/shrink, fixed 300px */
order: -1;            /* reorder items */
align-self: center;   /* override parent align */

📦 CSS Grid

/* Container */
display: grid;
grid-template-columns: repeat(3, 1fr);
grid-template-columns: repeat(auto-fill, minmax(250px, 1fr));
gap: 16px;
grid-template-areas:
  "header header"
  "sidebar main"
  "footer footer";

/* Item */
grid-column: 1 / 3;   /* span 2 columns */
grid-area: header;     /* use named area */

⚡ JavaScript Array Methods

const arr = [1, 2, 3, 4, 5];

arr.map(x => x * 2)           // [2, 4, 6, 8, 10]
arr.filter(x => x > 3)        // [4, 5]
arr.reduce((a, b) => a + b, 0) // 15
arr.find(x => x > 3)          // 4
arr.findIndex(x => x > 3)     // 3
arr.some(x => x > 4)          // true
arr.every(x => x > 0)         // true
arr.includes(3)               // true
arr.sort((a, b) => b - a)     // Descending
arr.flat()                     // Flatten nested

🔧 Fetch API

// GET
const res = await fetch("https://api.example.com/data");
const data = await res.json();

// POST
const res = await fetch("https://api.example.com/users", {
  method: "POST",
  headers: {"Content-Type": "application/json"},
  body: JSON.stringify({name: "Budi", age: 25})
});

// Error handling
if (!res.ok) throw new Error(`HTTP ${res.status}`);
