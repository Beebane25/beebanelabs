🏗️ StatelessWidget vs StatefulWidget

// StatelessWidget - tidak punya state
class MyStatelessWidget extends StatelessWidget {
  final String title;
  const MyStatelessWidget({required this.title});

  @override
  Widget build(BuildContext context) {
    return Text(title);
  }
}

// StatefulWidget - punya state yang bisa berubah
class MyStatefulWidget extends StatefulWidget {
  @override
  _MyStatefulWidgetState createState() => _MyStatefulWidgetState();
}

class _MyStatefulWidgetState extends State<MyStatefulWidget> {
  int _count = 0;

  @override
  Widget build(BuildContext context) {
    return Column(children: [
      Text('Count: $_count'),
      ElevatedButton(
        onPressed: () => setState(() => _count++),
        child: Text('Increment'),
      ),
    ]);
  }
}

🌐 HTTP Requests

import 'package:http/http.dart' as http;
import 'dart:convert';

// GET
Future<List<User>> getUsers() async {
  final response = await http.get(
    Uri.parse('https://api.example.com/users'),
    headers: {'Authorization': 'Bearer $token'},
  );
  if (response.statusCode == 200) {
    List data = json.decode(response.body);
    return data.map((e) => User.fromJson(e)).toList();
  }
  throw Exception('Failed to load users');
}

// POST
Future<User> createUser(User user) async {
  final response = await http.post(
    Uri.parse('https://api.example.com/users'),
    headers: {'Content-Type': 'application/json'},
    body: json.encode(user.toJson()),
  );
  return User.fromJson(json.decode(response.body));
}

📦 JSON Parsing

// Model class
class User {
  final int id;
  final String name;
  final String email;

  User({required this.id, required this.name, required this.email});

  factory User.fromJson(Map<String, dynamic> json) {
    return User(
      id: json['id'],
      name: json['name'],
      email: json['email'],
    );
  }

  Map<String, dynamic> toJson() {
    return {'id': id, 'name': name, 'email': email};
  }
}

// Parsing JSON
final user = User.fromJson(json.decode(responseBody));
