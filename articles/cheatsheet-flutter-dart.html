5. StatefulWidget

Widget dengan state yang bisa berubah selama lifecycle.

class CounterPage extends StatefulWidget {
  const CounterPage({super.key});

  @override
  State<CounterPage> createState() => _CounterPageState();
}

class _CounterPageState extends State<CounterPage> {
  int _counter = 0;

  @override
  void initState() {
    super.initState();
    // Inisialisasi data
  }

  void _increment() {
    setState(() {
      _counter++;
    });
  }

  @override
  void dispose() {
    // Cleanup controller, listener, dll
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(child: Text('$_counter', style: TextStyle(fontSize: 48))),
      floatingActionButton: FloatingActionButton(
        onPressed: _increment,
        child: Icon(Icons.add),
      ),
    );
  }
}

8. setState & State Management

// setState - cara dasar update UI
setState(() {
  _counter++;
  _isLoading = true;
});

// Best practice: jangan lakukan async work di dalam setState
Future<void> _loadData() async {
  setState(() => _isLoading = true);
  try {
    final data = await fetchData();
    setState(() {
      _items = data;
      _isLoading = false;
    });
  } catch (e) {
    setState(() => _isLoading = false);
  }
}

9. Provider Basics

// 1. Buat ChangeNotifier
class CartProvider extends ChangeNotifier {
  final List<String> _items = [];
  List<String> get items => _items;
  int get count => _items.length;

  void add(String item) {
    _items.add(item);
    notifyListeners();
  }

  void remove(String item) {
    _items.remove(item);
    notifyListeners();
  }
}

// 2. Daftarkan di atas widget tree
MultiProvider(
  providers: [
    ChangeNotifierProvider(create: (_) => CartProvider()),
    ChangeNotifierProvider(create: (_) => AuthProvider()),
  ],
  child: MyApp(),
)

// 3. Akses di widget
// Consumer pattern
Consumer<CartProvider>(
  builder: (context, cart, child) {
    return Text('Items: ${cart.count}');
  },
)

// context.read (one-time read, tidak rebuild)
final cart = context.read<CartProvider>();
cart.add('Produk A');

// context.watch (listen & rebuild)
final count = context.watch<CartProvider>().count;

// Selector (rebuild hanya jika value berubah)
Selector<CartProvider, int>(
  selector: (_, cart) => cart.count,
  builder: (_, count, __) => Text('$count items'),
)

10. HTTP Requests

Menggunakan http package

import 'package:http/http.dart' as http;
import 'dart:convert';

// GET
Future<List<User>> getUsers() async {
  final response = await http.get(
    Uri.parse('https://api.example.com/users'),
    headers: {'Authorization': 'Bearer $token'},
  );
  if (response.statusCode == 200) {
    final List data = jsonDecode(response.body);
    return data.map((e) => User.fromJson(e)).toList();
  }
  throw Exception('Gagal memuat data');
}

// POST
Future<User> createUser(String name, String email) async {
  final response = await http.post(
    Uri.parse('https://api.example.com/users'),
    headers: {'Content-Type': 'application/json'},
    body: jsonEncode({'name': name, 'email': email}),
  );
  if (response.statusCode == 201) {
    return User.fromJson(jsonDecode(response.body));
  }
  throw Exception('Gagal membuat user');
}

// PUT / DELETE
await http.put(Uri.parse('...'), body: jsonEncode(data));
await http.delete(Uri.parse('...'));

Menggunakan dio (alternatif)

import 'package:dio/dio.dart';

final dio = Dio(BaseOptions(

  connectTimeout: Duration(seconds: 5),

// Interceptor
dio.interceptors.add(LogInterceptor(responseBody: true));

// GET
final response = await dio.get('/users');
final users = response.data;

// POST dengan FormData
final formData = FormData.fromMap({
  'name': 'Budi',

await dio.post('/users', data: formData);

import 'dart:convert';

// Decode JSON string → Map/List
final Map<String, dynamic> map = jsonDecode('{"name":"Budi"}');

// Model class dengan fromJson / toJson
class User {
  final int id;
  final String name;
  final String email;

  User({required this.id, required this.name, required this.email});

  factory User.fromJson(Map<String, dynamic> json) {
    return User(
      id: json['id'] as int,
      name: json['name'] as String,
      email: json['email'] as String,
    );
  }

  Map<String, dynamic> toJson() => {
    'id': id,
    'name': name,
    'email': email,
  };
}

// Parsing list of objects
List<User> users = (jsonDecode(responseBody) as List)
    .map((e) => User.fromJson(e))
    .toList();

// Future async/await
Future<String> fetchData() async {
  final response = await http.get(Uri.parse('https://api.example.com/data'));
  return response.body;
}

// Error handling
Future<void> loadData() async {
  try {
    final data = await fetchData();
    print(data);
  } on SocketException {
    print('Tidak ada koneksi internet');
  } on HttpException catch (e) {
    print('HTTP error: $e');
  } catch (e) {
    print('Error: $e');
  } finally {
    print('Selesai');
  }
}

// FutureBuilder di widget
FutureBuilder<String>(
  future: fetchData(),
  builder: (context, snapshot) {
    if (snapshot.connectionState == ConnectionState.waiting) {
      return CircularProgressIndicator();
    }
    if (snapshot.hasError) {
      return Text('Error: ${snapshot.error}');
    }
    return Text(snapshot.data ?? '');
  },
)

// StreamBuilder
StreamBuilder<List<Message>>(
  stream: messageStream(),
  builder: (context, snapshot) {
    if (!snapshot.hasData) return CircularProgressIndicator();
    return ListView(
      children: snapshot.data!.map((m) => Text(m.body)).toList(),
    );
  },
)

// Parallel futures
final results = await Future.wait([
  fetchUsers(),
  fetchProducts(),
  fetchOrders(),
]);
