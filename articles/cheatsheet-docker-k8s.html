1. Docker CLI Commands

Image Management

# Build image
docker build -t namaimage:tag .
docker build -f Dockerfile.prod -t app:prod .

# List images
docker images
docker image ls

# Pull & Push
docker pull nginx:latest
docker push registry/namaimage:tag

# Tag image
docker tag app:latest registry.example.com/app:v1

# Remove
docker rmi namaimage:tag
docker image prune          # hapus dangling images
docker image prune -a       # hapus semua unused images

# Inspect
docker image inspect namaimage
docker history namaimage:tag

Container Management

# Run container
docker run -d -p 8080:80 --name web nginx
docker run -it ubuntu /bin/bash           # interaktif
docker run -e APP_ENV=production app      # environment variable
docker run -v /host/path:/container/path  # volume mount
docker run --network mynet app            # custom network
docker run --restart unless-stopped app   # restart policy

# List containers
docker ps                   # running only
docker ps -a                # all containers
docker ps -q                # quiet (IDs only)

# Start / Stop / Restart
docker start container_name
docker stop container_name
docker restart container_name

# Logs & Exec
docker logs container_name
docker logs -f container_name             # follow (tail)
docker logs --tail 100 container_name     # last 100 lines
docker exec -it container_name /bin/bash  # masuk ke container

# Remove
docker rm container_name
docker container prune                    # hapus stopped containers
docker rm -f container_name               # force remove

Docker Networking

# Network commands
docker network ls
docker network create mynet
docker network create --driver bridge mybridge
docker network connect mynet container_name
docker network disconnect mynet container_name
docker network inspect mynet
docker network rm mynet

Docker Volumes

# Volume commands
docker volume create mydata
docker volume ls
docker volume inspect mydata
docker volume rm mydata
docker volume prune

# Mount types
docker run -v mydata:/app/data app            # named volume
docker run -v $(pwd)/src:/app/src app         # bind mount
docker run --mount type=volume,src=mydata,dst=/app/data app

2. Dockerfile Syntax

# Multi-stage build example
# ---- Stage 1: Build ----
FROM node:20-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production
COPY . .
RUN npm run build

# ---- Stage 2: Production ----
FROM node:20-alpine
WORKDIR /app
RUN addgroup -g 1001 -S appgroup && \
    adduser -S appuser -u 1001
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/package.json ./
USER appuser
EXPOSE 3000
HEALTHCHECK --interval=30s --timeout=3s \
  CMD wget -qO- http://localhost:3000/health || exit 1
CMD ["node", "dist/server.js"]

3. Docker Compose

# docker-compose.yml
version: "3.9"

services:
  web:
    build:
      context: .
      dockerfile: Dockerfile
    ports:
      - "3000:3000"
    environment:
      - NODE_ENV=production
      - DB_HOST=db
    env_file:
      - .env
    volumes:
      - ./uploads:/app/uploads
    depends_on:
      db:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - appnet

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: myapp
      POSTGRES_USER: admin
      POSTGRES_PASSWORD: secret
    volumes:
      - pgdata:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U admin"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - appnet

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    networks:
      - appnet

volumes:
  pgdata:

networks:
  appnet:
    driver: bridge

Perintah Docker Compose

# Jalankan semua service
docker compose up -d

# Build & jalankan
docker compose up -d --build

# Lihat status
docker compose ps

# Logs
docker compose logs -f web

# Stop & hapus
docker compose down
docker compose down -v         # hapus juga volumes

# Scale service
docker compose up -d --scale web=3

# Execute command
docker compose exec web bash
docker compose run --rm web npm test

4. Docker Best Practices

PraktikAlasan
Gunakan multi-stage buildKurangi ukuran image final
Gunakan Alpine/Slim base imagesImage lebih kecil, attack surface minimal
Jangan jalankan sebagai rootKeamanan: gunakan USER directive
Gunakan .dockerignoreBuild context lebih cepat & bersih
Minimalkan layer RUNKurangi ukuran image (gabungkan perintah)
Copy dependency file duluanManfaatkan Docker layer cache
Gunakan HEALTHCHECKDocker bisa monitor status container
Pin versi base imageReproducible builds
Jangan simpan secrets di imageGunakan ENV/runtime injection
Gunakan docker scanDeteksi vulnerability

6. kubectl Commands

Dasar

# Info cluster
kubectl cluster-info
kubectl get nodes
kubectl get nodes -o wide

# Get resources
kubectl get pods
kubectl get pods -A                          # semua namespace
kubectl get pods -n kube-system              # namespace tertentu
kubectl get pods -o wide                     # detail tambahan
kubectl get pods -l app=web                  # filter label
kubectl get pods --field-selector=status.phase=Running
kubectl get all                              # semua resource
kubectl get svc,deploy,pods                  # beberapa tipe

# Describe (detail)
kubectl describe pod nama_pod
kubectl describe service nama_svc

# Logs
kubectl logs nama_pod
kubectl logs -f nama_pod                     # follow
kubectl logs nama_pod -c container_name      # multi-container
kubectl logs nama_pod --tail=50              # 50 baris terakhir

# Exec
kubectl exec -it nama_pod -- /bin/bash
kubectl exec nama_pod -- ls /app

# Port forwarding
kubectl port-forward svc/nama_svc 8080:80
kubectl port-forward pod/nama_pod 3000:3000

# Copy file
kubectl cp nama_pod:/app/logs.txt ./logs.txt

Deployment Management

# Create deployment
kubectl create deployment web --image=nginx:latest --replicas=3

# Scale
kubectl scale deployment web --replicas=5

# Update image
kubectl set image deployment/web web=nginx:1.25

# Rollout
kubectl rollout status deployment/web
kubectl rollout history deployment/web
kubectl rollout undo deployment/web              # rollback
kubectl rollout undo deployment/web --to-revision=2

# Delete
kubectl delete pod nama_pod
kubectl delete deployment nama_deploy
kubectl delete -f manifest.yaml
kubectl delete pods --all                        # semua pod

Labels & Annotations

# Label
kubectl label pod nama_pod env=production
kubectl label pod nama_pod env-                 # hapus label

# Annotation
kubectl annotate pod nama_pod note="ini pod testing"

7. YAML Manifests

Deployment

apiVersion: apps/v1
kind: Deployment
metadata:
  name: web-app
  labels:
    app: web
spec:
  replicas: 3
  selector:
    matchLabels:
      app: web
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  template:
    metadata:
      labels:
        app: web
    spec:
      containers:
        - name: web
          image: nginx:1.25-alpine
          ports:
            - containerPort: 80
          env:
            - name: APP_ENV
              value: "production"
            - name: DB_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: db-secret
                  key: password
          resources:
            requests:
              cpu: "100m"
              memory: "128Mi"
            limits:
              cpu: "500m"
              memory: "256Mi"
          livenessProbe:
            httpGet:
              path: /health
              port: 80
            initialDelaySeconds: 10
            periodSeconds: 15
          readinessProbe:
            httpGet:
              path: /ready
              port: 80
            initialDelaySeconds: 5
            periodSeconds: 5

Service

apiVersion: v1
kind: Service
metadata:
  name: web-service
spec:
  type: ClusterIP        # ClusterIP | NodePort | LoadBalancer
  selector:
    app: web
  ports:
    - protocol: TCP
      port: 80            # service port
      targetPort: 80      # container port

Ingress

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: web-ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  rules:
    - host: app.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: web-service
                port:
                  number: 80

8. Rolling Updates

# Update image secara rolling
kubectl set image deployment/web web=nginx:1.26

# Monitor rollout
kubectl rollout status deployment/web --watch

# Lihat history
kubectl rollout history deployment/web

# Rollback ke revisi sebelumnya
kubectl rollout undo deployment/web

# Rollback ke revisi spesifik
kubectl rollout undo deployment/web --to-revision=3

# Pause & resume rollout
kubectl rollout pause deployment/web
kubectl rollout resume deployment/web

# Restart deployment (rolling restart)
kubectl rollout restart deployment/web

Strategy Options

StrategyKeteranganUse Case
RollingUpdateUpdate bertahapDefault, zero-downtime
RecreateHapus semua, buat baruApp tidak support 2 versi bersamaan
maxSurge: 1Max 1 pod baru saat updateControl resource usage
maxUnavailable: 0Selalu ada pod availableZero-downtime update

9. Resource Limits

# Di dalam pod spec
resources:
  requests:
    cpu: "250m"        # 0.25 CPU core
    memory: "256Mi"    # 256 MB RAM
  limits:
    cpu: "1000m"       # 1 CPU core (maksimal)
    memory: "512Mi"    # 512 MB RAM (maksimal)

10. ConfigMaps & Secrets

ConfigMap

# Buat dari literal
kubectl create configmap app-config \
  --from-literal=APP_ENV=production \
  --from-literal=LOG_LEVEL=info

# Buat dari file
kubectl create configmap nginx-config --from-file=nginx.conf

# YAML manifest
apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  APP_ENV: "production"
  LOG_LEVEL: "info"
  config.json: |
    {"key": "value"}

# Gunakan di Pod
env:
  - name: APP_ENV
    valueFrom:
      configMapKeyRef:
        name: app-config
        key: APP_ENV

# Atau mount sebagai volume
volumes:
  - name: config-volume
    configMap:
      name: app-config

Secrets

# Buat secret
kubectl create secret generic db-secret \
  --from-literal=username=admin \
  --from-literal=password=s3cretP@ss

# TLS secret
kubectl create secret tls my-tls \
  --cert=cert.pem --key=key.pem

# Docker registry secret
kubectl create secret docker-registry regcred \
  --docker-server=registry.example.com \
  --docker-username=user \
  --docker-password=pass

# YAML
apiVersion: v1
kind: Secret
metadata:
  name: db-secret
type: Opaque
data:
  username: YWRtaW4=          # base64 encoded
  password: czNjcmV0UEBzcw==

# Decode base64
echo "YWRtaW4=" | base64 -d

# Encode base64
echo -n "admin" | base64

# Gunakan di Pod
env:
  - name: DB_USER
    valueFrom:
      secretKeyRef:
        name: db-secret
        key: username

11. Helm Basics

Helm Commands

# Tambah repo
helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update

# Cari chart
helm search repo nginx
helm search hub wordpress

# Install
helm install my-release bitnami/nginx
helm install my-release bitnami/nginx \
  --set replicaCount=3 \
  --set service.type=LoadBalancer

# Install dengan values file
helm install my-release bitnami/nginx -f values.yaml

# Upgrade
helm upgrade my-release bitnami/nginx --set replicaCount=5

# Rollback
helm rollback my-release 1

# List & status
helm list
helm list -A                     # semua namespace
helm status my-release

# History & values
helm history my-release
helm get values my-release

# Uninstall
helm uninstall my-release

values.yaml Example

# values.yaml
replicaCount: 3
image:
  repository: nginx
  tag: "1.25-alpine"
  pullPolicy: IfNotPresent
service:
  type: ClusterIP
  port: 80
resources:
  requests:
    cpu: 100m
    memory: 128Mi
  limits:
    cpu: 500m
    memory: 256Mi
ingress:
  enabled: true
  host: app.example.com
autoscaling:
  enabled: true
  minReplicas: 2
  maxReplicas: 10
  targetCPU: 80

Buat Chart Sendiri

# Buat chart baru
helm create my-chart

# Struktur
my-chart/
  Chart.yaml          # metadata chart
  values.yaml         # default values
  charts/             # dependency charts
  templates/
    deployment.yaml
    service.yaml
    ingress.yaml
    _helpers.tpl      # template helpers
    NOTES.txt         # post-install notes

# Package & lint
helm lint my-chart
helm package my-chart
helm template my-chart --debug    # render tanpa install
