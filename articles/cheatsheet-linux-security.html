Cheatsheet Linux Security: Command Line, Firewall, dan Hardening

Referensi cepat command Linux untuk keamanan: iptables, SSH hardening, user management, dan network scanning.

📅 3 Juli 2026 📖 10 menit baca Pemula

🔒 iptables / nftables

# iptables basic rules
iptables -A INPUT -p tcp --dport 22 -j ACCEPT    # Allow SSH
iptables -A INPUT -p tcp --dport 80 -j ACCEPT    # Allow HTTP
iptables -A INPUT -p tcp --dport 443 -j ACCEPT   # Allow HTTPS
iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT
iptables -A INPUT -j DROP                         # Drop sisanya

# Block IP
iptables -A INPUT -s 192.168.1.100 -j DROP

# Rate limiting SSH (anti brute-force)
iptables -A INPUT -p tcp --dport 22 -m recent --set --name SSH
iptables -A INPUT -p tcp --dport 22 -m recent --update --seconds 60 --hitcount 4 --name SSH -j DROP

# Save rules
iptables-save > /etc/iptables/rules.v4

🔑 SSH Hardening

# /etc/ssh/sshd_config
Port 2222                    # Non-default port
PermitRootLogin no           # Disable root login
PasswordAuthentication no    # Key-only auth
MaxAuthTries 3               # Limit attempts
AllowUsers admin deploy      # Whitelist users
Protocol 2                   # SSH2 only
ClientAliveInterval 300
ClientAliveCountMax 2

# Restart SSH
sudo systemctl restart sshd

👤 User Management

# Create user with sudo
useradd -m -G sudo -s /bin/bash newuser
passwd newuser

# Lock/unlock account
usermod -L username          # Lock
usermod -U username          # Unlock

# Check last logins
lastlog
last -n 10
who
w

chown -R www-data:www-data /var/www/
find / -perm -4000 -type f   # Find SUID files

🔍 Network Scanning

# nmap
nmap -sV -sC target.com              # Service detection
nmap -sS -O target.com               # SYN scan + OS detection
nmap -p 1-65535 target.com           # Full port scan
nmap --script vuln target.com        # Vulnerability scan
nmap -sn 192.168.1.0/24              # Ping sweep

# Netstat / ss
ss -tulnp                            # Listening ports
ss -s                                # Connection stats
netstat -antp                        # All connections

📋 Log Analysis

# Auth logs
grep "Failed password" /var/log/auth.log | wc -l
grep "Failed password" /var/log/auth.log | awk '{print $11}' | sort | uniq -c | sort -rn

# Journal
journalctl -u sshd --since "1 hour ago"
journalctl -p err --since today

# Real-time monitoring
tail -f /var/log/auth.log | grep --color "Failed"

🛡️ fail2ban

# /etc/fail2ban/jail.local
[sshd]
enabled = true
port = 2222
maxretry = 3
bantime = 3600
findtime = 600

# Commands
fail2ban-client status sshd
fail2ban-client set sshd unbanip 1.2.3.4
