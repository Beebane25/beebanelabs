🔧 Arduino IDE Commands

// Setup ESP32 WiFi
#include <WiFi.h>
const char* ssid = "NamaWiFi";
const char* pass = "PasswordWiFi";
WiFi.begin(ssid, pass);
while (WiFi.status() != WL_CONNECTED) {
  delay(500);
  Serial.print(".");
}
Serial.println(WiFi.localIP());
// ESP32 GPIO Control
pinMode(2, OUTPUT);       // Set GPIO 2 sebagai output
digitalWrite(2, HIGH);    // Nyalakan LED
analogWrite(25, 128);     // PWM 50% pada GPIO 25
int val = analogRead(34); // Baca ADC dari GPIO 34

📡 MQTT Quick Reference

#include <PubSubClient.h>
WiFiClient espClient;
PubSubClient client(espClient);

client.setServer("broker.mqtt-dashboard.com", 1883);
client.connect("ESP32Client");
client.publish("iot/sensor", "hello");
client.subscribe("iot/control");
client.setCallback(callback);

void callback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) message += (char)payload[i];
  Serial.println(message);
}

🐍 MicroPython Basics

# MicroPython ESP32
import machine
import network
from umqtt.simple import MQTTClient

# WiFi
sta = network.WLAN(network.STA_IF)
sta.active(True)
sta.connect("WiFi", "password")

# GPIO
led = machine.Pin(2, machine.Pin.OUT)
led.on()   # Nyalakan
led.off()  # Matikan

# ADC
adc = machine.ADC(machine.Pin(34))
adc.atten(machine.ADC.ATTN_11DB)
value = adc.read()  # 0-4095
