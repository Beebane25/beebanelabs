Rumus Subnetting

Jumlah host = 2^n - 2    (n = jumlah bit host)
Jumlah subnet = 2^n      (n = jumlah bit subnet)
Block size = 256 - subnet mask octet

Contoh: 192.168.1.0/26
  → Bit host = 32 - 26 = 6
  → Jumlah host = 2^6 - 2 = 62
  → Block size = 256 - 192 = 64
  → Subnet: .0, .64, .128, .192
  → Range .0 subnet: 192.168.1.1 - 192.168.1.62 (broadcast .63)

4. MikroTik CLI Basics

Akses & Navigasi

# Akses via SSH
ssh admin@192.168.88.1

# Akses via Winbox
# Download winbox dari mikrotik.com, konek ke IP router

# Navigasi menu (seperti cd)
/interface           # masuk ke menu interface
/ip address          # masuk ke menu IP address
/system              # masuk ke menu system

# Lihat konfigurasi
/interface print
/ip address print
/ip route print
/system resource print

# Shortcut
/interface print detail
/ip address print where address~"192.168"
/interface print stats

# Export konfigurasi
/export compact
/export file=backup-config

System Management

# Identity & hostname
/system identity set name=RouterKantor

# Password admin
/user set admin password=NewSecurePass123

# Tambah user baru
/user add name=operator password=OpPass group=full

# NTP / waktu
/system ntp client set enabled=yes
/system ntp client servers add address=pool.ntp.org

# DNS
/ip dns set servers=8.8.8.8,8.8.4.4 allow-remote-requests=yes

# Backup & restore
/system backup save name=backup-before
/system backup load name=backup-before

# Update RouterOS
/system package update check-for-updates
/system package update install

# Reboot & shutdown
/system reboot
/system shutdown

5. Interface Management

# Lihat semua interface
/interface print

# Set IP address
/ip address add address=192.168.1.1/24 interface=ether1
/ip address add address=10.0.0.1/24 interface=ether2

# Rename interface
/interface set ether1 name=WAN
/interface set ether2 name=LAN

# Enable/disable interface
/interface disable ether3
/interface enable ether3

# Bridge (gabung beberapa port)
/interface bridge add name=bridge1
/interface bridge port add bridge=bridge1 interface=ether2
/interface bridge port add bridge=bridge1 interface=ether3
/interface bridge port add bridge=bridge1 interface=ether4

# DHCP Client (untuk WAN - dapat IP dari ISP)
/ip dhcp-client add interface=WAN disabled=no

# Static route
/ip route add dst-address=0.0.0.0/0 gateway=10.0.0.1
/ip route add dst-address=172.16.0.0/16 gateway=192.168.1.254

6. Firewall Rules

NAT (Masquerade) - Internet Sharing

# Masquerade (NAT untuk akses internet dari LAN)
/ip firewall nat add chain=srcnat out-interface=WAN action=masquerade

# Port forwarding (contoh: forward port 80 ke server internal)
/ip firewall nat add chain=dstnat dst-port=80 protocol=tcp \
  in-interface=WAN action=dst-nat to-addresses=192.168.1.100 to-ports=80

# Port forwarding SSH
/ip firewall nat add chain=dstnat dst-port=2222 protocol=tcp \
  in-interface=WAN action=dst-nat to-addresses=192.168.1.10 to-ports=22

Filter Rules

# Drop semua yang tidak diizinkan (basic firewall)
/ip firewall filter add chain=input connection-state=established action=accept
/ip firewall filter add chain=input connection-state=related action=accept
/ip firewall filter add chain=input connection-state=invalid action=drop

# Izinkan Winbox dari LAN
/ip firewall filter add chain=input src-address=192.168.1.0/24 dst-port=8291 \
  protocol=tcp action=accept

# Izinkan SSH dari LAN
/ip firewall filter add chain=input src-address=192.168.1.0/24 dst-port=22 \
  protocol=tcp action=accept

# Izinkan ICMP (ping)
/ip firewall filter add chain=input protocol=icmp action=accept

# Drop semua input lain (taruh di akhir)
/ip firewall filter add chain=input action=drop comment="Drop all other input"

# Block website/IP tertentu
/ip firewall filter add chain=forward dst-address=123.45.67.89 action=drop

# Limit koneksi (anti DDoS)
/ip firewall filter add chain=forward connection-limit=50,32 action=drop

# Block P2P
/ip firewall layer7-protocol add name=block-p2p regexp="torrent"
/ip firewall filter add chain=forward layer7-protocol=block-p2p action=drop

7. DHCP Setup

# Setup DHCP Server
/ip pool add name=dhcp-pool ranges=192.168.1.100-192.168.1.200

/ip dhcp-server add name=dhcp1 interface=LAN \
  address-pool=dhcp-pool lease-time=1d disabled=no

/ip dhcp-server network add address=192.168.1.0/24 \
  gateway=192.168.1.1 dns-server=8.8.8.8,8.8.4.4 \
  domain=local.network

# Static DHCP lease (IP tetap untuk device tertentu)
/ip dhcp-server lease add mac-address=AA:BB:CC:DD:EE:FF \
  address=192.168.1.50 server=dhcp1 comment="Server Web"

# DHCP Client (router sebagai client, dapat IP dari ISP)
/ip dhcp-client add interface=WAN disabled=no

# Lihat lease aktif
/ip dhcp-server lease print

# Lihat DHCP server status
/ip dhcp-server print

8. VPN Configuration

PPTP VPN

# Server PPTP
/interface pptp-server server set enabled=yes
/ppp secret add name=user1 password=pass123 \
  local-address=10.10.10.1 remote-address=10.10.10.100 \
  service=pptp

/ip firewall nat add chain=srcnat src-address=10.10.10.0/24 \
  out-interface=WAN action=masquerade

L2TP/IPsec VPN

# Server L2TP/IPsec
/ip ipsec proposal set [ find default=yes ] \
  auth-algorithms=sha256 enc-algorithms=aes-256-cbc

/ppp profile add name=l2tp-profile \
  local-address=10.10.10.1 remote-address=10.10.10.0/24 \
  use-encryption=yes

/interface l2tp-server server set enabled=yes \
  default-profile=l2tp-profile \
  ipsec-secret=MyIPsecKey \
  use-ipsec=yes

/ppp secret add name=vpnuser password=vpnpass service=l2tp \
  profile=l2tp-profile

WireGuard VPN (RouterOS 7+)

# Buat WireGuard interface
/interface wireguard add name=wg1 listen-port=13231

# Generate keys (otomatis saat buat interface)
# Atau manual:
/interface wireguard peers add interface=wg1 \
  public-key="client_public_key_here" \
  allowed-address=10.20.30.2/32

# Tambah IP
/ip address add address=10.20.30.1/24 interface=wg1

# Firewall - izinkan WireGuard port
/ip firewall filter add chain=input dst-port=13231 \
  protocol=udp action=accept

9. VLAN Setup

# Buat VLAN interface
/interface vlan add name=vlan10 vlan-id=10 interface=ether1
/interface vlan add name=vlan20 vlan-id=20 interface=ether1
/interface vlan add name=vlan30 vlan-id=30 interface=ether1

# Set IP address untuk VLAN
/ip address add address=192.168.10.1/24 interface=vlan10
/ip address add address=192.168.20.1/24 interface=vlan20
/ip address add address=192.168.30.1/24 interface=vlan30

# Bridge per VLAN (jika switch tidak support hardware VLAN)
/interface bridge add name=bridge-vlan10
/interface bridge port add bridge=bridge-vlan10 interface=vlan10
/interface bridge port add bridge=bridge-vlan10 interface=ether2

/interface bridge add name=bridge-vlan20
/interface bridge port add bridge=bridge-vlan20 interface=vlan20
/interface bridge port add bridge=bridge-vlan20 interface=ether3

# DHCP per VLAN
/ip pool add name=pool-vlan10 ranges=192.168.10.100-192.168.10.200
/ip dhcp-server add name=dhcp-vlan10 interface=bridge-vlan10 \
  address-pool=pool-vlan10 disabled=no
/ip dhcp-server network add address=192.168.10.0/24 \
  gateway=192.168.10.1 dns-server=8.8.8.8

/ip pool add name=pool-vlan20 ranges=192.168.20.100-192.168.20.200
/ip dhcp-server add name=dhcp-vlan20 interface=bridge-vlan20 \
  address-pool=pool-vlan20 disabled=no
/ip dhcp-server network add address=192.168.20.0/24 \
  gateway=192.168.20.1 dns-server=8.8.8.8

# Inter-VLAN routing (otomatis karena semua VLAN di router sama)
# Block antar VLAN (jika perlu isolasi)
/ip firewall filter add chain=forward \
  src-address=192.168.10.0/24 dst-address=192.168.20.0/24 action=drop

10. Bandwidth Management

Simple Queue

# Limit bandwidth per user
/queue simple add name=User1 target=192.168.1.100/32 \
  max-limit=5M/10M comment="Download 5Mbps, Upload 10Mbps"

# Limit per subnet
/queue simple add name=Staff target=192.168.1.0/24 \
  max-limit=20M/40M

# Prioritas (kecil = prioritas tinggi, 1-8)
/queue simple add name=VIP target=192.168.1.50/32 \
  max-limit=10M/20M priority=1/1

# Burst (pinjam bandwidth sementara)
/queue simple add name=BurstUser target=192.168.1.101/32 \
  max-limit=2M/5M burst-limit=10M/20M burst-threshold=1M/2M \
  burst-time=10s/10s

Queue Tree (HTB - advanced)

# Mark packet dulu
/ip firewall mangle add chain=forward src-address=192.168.1.0/24 \
  action=mark-connection new-connection-mark=lan-conn
/ip firewall mangle add chain=forward connection-mark=lan-conn \
  action=mark-packet new-packet-mark=lan-packet

# Buat parent queue
/queue tree add name=Download parent=global max-limit=50M
/queue tree add name=Upload parent=global max-limit=20M

# Child queue
/queue tree add name=User-DL parent=Download packet-mark=lan-packet \
  limit-at=1M max-limit=10M priority=4

PCQ (Per Connection Queue)

# PCQ untuk bandwidth fair sharing
/queue type add name=pcq-download kind=pcq \
  pcq-classifier=dst-address pcq-rate=2M
/queue type add name=pcq-upload kind=pcq \
  pcq-classifier=src-address pcq-rate=1M

# Mark packet
/ip firewall mangle add chain=forward src-address=192.168.1.0/24 \
  action=mark-packet new-packet-mark=client-up
/ip firewall mangle add chain=forward dst-address=192.168.1.0/24 \
  action=mark-packet new-packet-mark=client-down

# Terapkan queue tree
/queue tree add name=Upload parent=global \
  packet-mark=client-up queue=pcq-upload
/queue tree add name=Download parent=global \
  packet-mark=client-down queue=pcq-download

11. Troubleshooting Commands

Diagnostik Jaringan

# Ping
/ping 8.8.8.8
/ping 8.8.8.8 count=10 size=1400

# Traceroute
/tool traceroute 8.8.8.8

# DNS lookup
/tool dns-check domain=google.com

# Bandwidth test
/tool bandwidth-test address=192.168.1.1 duration=10s

# Speed test (RouterOS 7+)
/tool speed-test address=192.168.1.1

# Scan network (neighbors)
/ip neighbor print
/ip arp print

# Torch (monitor traffic real-time)
/tool torch interface=WAN
/tool torch interface=LAN src-address=192.168.1.100

# Profile CPU
/system resource cpu print
/system resource monitor

# Lihat log
/log print
/log print where topics~"firewall"
/log print where topics~"dhcp"

# Connection tracking
/ip firewall connection tracking print
/ip firewall connection print where src-address~"192.168.1"

Monitoring

# Interface stats
/interface monitor-traffic interface=WAN

# Resource usage
/system resource print

# Health (temperatur, voltage)
/system health print

# Graphing
/tool graphing interface add interface=WAN
/tool graphing queue add
/tool graphing resource add
